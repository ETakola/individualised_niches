timestamp <- function() {
  format(Sys.time(), "%Y-%m-%d %H:%M:%S")
}

log_msg <- function(...) {
  msg <- paste0(...)
  line <- paste0("[", timestamp(), "] ", msg)
  message(line)
  cat(line, "\n", file = log_file, append = TRUE)
}

add_warning <- function(section, msg) {
  .workflow_warnings <<- dplyr::bind_rows(
    .workflow_warnings,
    tibble::tibble(
      time = timestamp(),
      section = section,
      message = msg
    )
  )
  
  utils::write.csv(
    .workflow_warnings,
    warning_file,
    row.names = FALSE
  )
  
  log_msg("WARNING in ", section, ": ", msg)
}

add_error <- function(section, msg) {
  .workflow_errors <<- dplyr::bind_rows(
    .workflow_errors,
    tibble::tibble(
      time = timestamp(),
      section = section,
      message = msg
    )
  )
  
  utils::write.csv(
    .workflow_errors,
    error_file,
    row.names = FALSE
  )
  
  log_msg("ERROR in ", section, ": ", msg)
}

# IMPORTANT:
# This is the safe warning pattern.
# invokeRestart("muffleWarning") is only called inside withCallingHandlers(),
# where the muffle Warning restart actually exists.
safe_section <- function(section, expr) {
  expr_sub <- substitute(expr)
  
  tryCatch(
    withCallingHandlers(
      eval(expr_sub, envir = parent.frame()),
      warning = function(w) {
        add_warning(section, conditionMessage(w))
        invokeRestart("muffleWarning")
      }
    ),
    error = function(e) {
      add_error(section, conditionMessage(e))
      NULL
    }
  )
}

case_to_integer <- function(x) {
  if (is.logical(x)) return(as.integer(x))
  
  if (is.factor(x)) {
    x <- as.character(x)
  }
  
  if (is.character(x)) {
    out <- dplyr::case_when(
      tolower(x) %in% c("1", "true", "used", "realized", "realised") ~ 1L,
      tolower(x) %in% c("0", "false", "available", "potential") ~ 0L,
      TRUE ~ NA_integer_
    )
    return(out)
  }
  
  as.integer(x)
}

safe_name <- function(x) {
  x |>
    as.character() |>
    stringr::str_replace_all("[^A-Za-z0-9_]+", "_")
}

clean_weights <- function(w) {
  w <- as.numeric(w)
  w[!is.finite(w) | w < 0] <- 0
  
  if (sum(w, na.rm = TRUE) <= 0) {
    return(NULL)
  }
  
  w / sum(w, na.rm = TRUE)
}

get_hv_volume <- function(hv) {
  if (is.null(hv)) return(NA_real_)
  
  tryCatch(
    as.numeric(hypervolume::get_volume(hv)),
    error = function(e) {
      warning(paste("Could not extract hypervolume volume:", e$message))
      NA_real_
    }
  )
}

close_polygon <- function(df) {
  if (is.null(df) || nrow(df) < 3) return(NULL)
  
  if (!all(c("PC1", "PC2") %in% names(df))) return(NULL)
  if (any(!is.finite(df$PC1)) || any(!is.finite(df$PC2))) {
    df <- df |>
      dplyr::filter(is.finite(PC1), is.finite(PC2))
  }
  
  if (nrow(df) < 3) return(NULL)
  
  idx <- chull(df$PC1, df$PC2)
  out <- df[idx, , drop = FALSE]
  out <- out[c(seq_len(nrow(out)), 1), , drop = FALSE]
  out
}

resolve_boot_path <- function(base_dir, path) {
  if (length(path) == 0 || all(is.na(path))) {
    return(NA_character_)
  }
  
  path <- as.character(path)[1]
  
  if (dir.exists(path)) {
    return(normalizePath(path, mustWork = FALSE))
  }
  
  candidate <- file.path(base_dir, path)
  
  if (dir.exists(candidate)) {
    return(normalizePath(candidate, mustWork = FALSE))
  }
  
  NA_character_
}

prep_group_data <- function(df, group_name, vars, rsp_col, extra_cols = character()) {
  vars_present <- intersect(vars, names(df))
  
  if (length(vars_present) == 0) {
    warning(paste("No variables found for group:", group_name))
    return(NULL)
  }
  
  if (!rsp_col %in% names(df)) {
    warning(paste("Missing RSP column:", rsp_col))
    return(NULL)
  }
  
  extra_cols <- intersect(extra_cols, names(df))
  
  hv_base <- df |>
    dplyr::mutate(
      id = factor(id),
      case_num = case_to_integer(case_),
      niche_type = dplyr::if_else(case_num == 1L, "Realized", "Potential"),
      niche_type = factor(niche_type, levels = niche_levels),
      weight_raw = as.numeric(.data[[rsp_col]])
    ) |>
    dplyr::select(
      id,
      case_num,
      niche_type,
      dplyr::all_of(extra_cols),
      weight_raw,
      dplyr::all_of(vars_present)
    ) |>
    tidyr::drop_na(dplyr::all_of(vars_present)) |>
    dplyr::filter(
      !is.na(id),
      !is.na(case_num),
      !is.na(niche_type)
    )
  
  if (nrow(hv_base) == 0) {
    warning(paste("No complete rows after dropping missing environmental values for group:", group_name))
    return(NULL)
  }
  
  num_vars_grp <- vars_present[
    vapply(hv_base[vars_present], is.numeric, logical(1))
  ]
  
  fac_vars_grp <- vars_present[
    vapply(
      hv_base[vars_present],
      function(x) is.factor(x) || is.character(x),
      logical(1)
    )
  ]
  
  if (length(fac_vars_grp) > 0) {
    hv_base <- hv_base |>
      dplyr::mutate(dplyr::across(dplyr::all_of(fac_vars_grp), as.factor))
  }
  
  if (length(num_vars_grp) > 0) {
    num_vars_grp <- num_vars_grp[
      vapply(
        hv_base[num_vars_grp],
        function(x) {
          sx <- stats::sd(x, na.rm = TRUE)
          is.finite(sx) && sx > 0
        },
        logical(1)
      )
    ]
  }
  
  if (length(num_vars_grp) > 0) {
    hv_num <- hv_base |>
      dplyr::select(dplyr::all_of(num_vars_grp)) |>
      dplyr::mutate(
        dplyr::across(
          dplyr::everything(),
          ~ as.numeric(scale(.x))
        )
      )
  } else {
    hv_num <- data.frame(dummy = seq_len(nrow(hv_base)))[, 0, drop = FALSE]
  }
  
  if (length(fac_vars_grp) > 0) {
    hv_fac <- stats::model.matrix(
      ~ . - 1,
      data = hv_base |>
        dplyr::select(dplyr::all_of(fac_vars_grp))
    ) |>
      as.data.frame()
    
    names(hv_fac) <- make.names(names(hv_fac), unique = TRUE)
  } else {
    hv_fac <- data.frame(dummy = seq_len(nrow(hv_base)))[, 0, drop = FALSE]
  }
  
  hv_dat <- dplyr::bind_cols(
    hv_base |>
      dplyr::select(
        id,
        case_num,
        niche_type,
        dplyr::all_of(extra_cols),
        weight_raw
      ),
    hv_num,
    hv_fac
  )
  
  axis_df <- hv_dat |>
    dplyr::select(
      -id,
      -case_num,
      -niche_type,
      -dplyr::all_of(extra_cols),
      -weight_raw
    ) |>
    dplyr::mutate(dplyr::across(dplyr::everything(), as.numeric))
  
  hv_axes <- names(axis_df)[
    vapply(
      axis_df,
      function(x) {
        sx <- stats::sd(x, na.rm = TRUE)
        is.finite(sx) && sx > 0
      },
      logical(1)
    )
  ]
  
  if (length(hv_axes) == 0) {
    warning(paste("No usable hypervolume axes for group:", group_name))
    return(NULL)
  }
  
  hv_dat <- dplyr::bind_cols(
    hv_dat |>
      dplyr::select(
        id,
        case_num,
        niche_type,
        dplyr::all_of(extra_cols),
        weight_raw
      ),
    axis_df |>
      dplyr::select(dplyr::all_of(hv_axes))
  )
  
  axis_matrix <- hv_dat |>
    dplyr::select(dplyr::all_of(hv_axes)) |>
    as.data.frame()
  
  global_bw <- tryCatch(
    {
      hypervolume::estimate_bandwidth(axis_matrix)
    },
    error = function(e) {
      warning(paste("Bandwidth estimation failed for group", group_name, ":", e$message))
      NULL
    }
  )
  
  if (is.null(global_bw)) return(NULL)
  
  pca_vis <- NULL
  
  if (ncol(axis_matrix) >= 2 && nrow(axis_matrix) >= 3) {
    pca_vis <- tryCatch(
      stats::prcomp(axis_matrix, center = TRUE, scale. = FALSE),
      error = function(e) {
        warning(paste("PCA visualization failed for group", group_name, ":", e$message))
        NULL
      }
    )
  }
  
  axis_summary <- tibble::tibble(
    group = group_name,
    rsp_column = rsp_col,
    vars_present = paste(vars_present, collapse = ", "),
    hv_axes = paste(hv_axes, collapse = ", "),
    n_rows = nrow(hv_dat),
    n_axes = length(hv_axes),
    n_numeric_vars_used = length(num_vars_grp),
    n_factor_vars_used = length(fac_vars_grp)
  )
  
  list(
    group = group_name,
    rsp_col = rsp_col,
    vars_present = vars_present,
    hv_axes = hv_axes,
    hv_dat = hv_dat,
    global_bw = global_bw,
    pca_vis = pca_vis,
    axis_summary = axis_summary
  )
}

make_hv <- function(
    dat,
    hv_axes,
    global_bw,
    hv_name,
    group,
    id,
    niche_type,
    season = NA_character_,
    use_weights = TRUE
) {
  if (is.null(dat) || nrow(dat) == 0) {
    add_skipped_hv(
      group = group,
      id = id,
      season = season,
      niche_type = niche_type,
      hv_name = hv_name,
      n_points = 0,
      n_axes = length(hv_axes),
      reason = "No rows"
    )
    return(NULL)
  }
  
  x <- dat |>
    dplyr::select(dplyr::all_of(hv_axes)) |>
    as.data.frame()
  
  x <- x |>
    dplyr::mutate(dplyr::across(dplyr::everything(), as.numeric))
  
  complete_finite <- stats::complete.cases(x) &
    apply(as.matrix(x), 1, function(z) all(is.finite(z)))
  
  if (!all(complete_finite)) {
    x <- x[complete_finite, , drop = FALSE]
    dat <- dat[complete_finite, , drop = FALSE]
  }
  
  min_n <- max(20, length(hv_axes) * min_points_factor)
  
  if (nrow(x) < min_n) {
    reason <- paste0(
      "Too few points: ",
      nrow(x),
      " rows for ",
      length(hv_axes),
      " axes. Needed at least ",
      min_n,
      "."
    )
    
    add_skipped_hv(
      group = group,
      id = id,
      season = season,
      niche_type = niche_type,
      hv_name = hv_name,
      n_points = nrow(x),
      n_axes = length(hv_axes),
      reason = reason
    )
    
    warning(paste("Skipped", hv_name, "because", reason))
    return(NULL)
  }
  
  w <- NULL
  
  if (isTRUE(use_weights)) {
    w <- clean_weights(dat$weight_raw)
  }
  
  tryCatch(
    {
      hypervolume::hypervolume_gaussian(
        data = x,
        name = hv_name,
        weight = w,
        kde.bandwidth = global_bw,
        samples.per.point = samples.per.point,
        quantile.requested = hv_quantile,
        quantile.requested.type = "probability",
        verbose = FALSE
      )
    },
    error = function(e) {
      add_skipped_hv(
        group = group,
        id = id,
        season = season,
        niche_type = niche_type,
        hv_name = hv_name,
        n_points = nrow(x),
        n_axes = length(hv_axes),
        reason = paste("Hypervolume failed:", e$message)
      )
      
      warning(paste("Failed", hv_name, ":", e$message))
      NULL
    }
  )
}

get_overlap_observed <- function(hv1, hv2) {
  if (is.null(hv1) || is.null(hv2)) {
    return(
      tibble::tibble(
        jaccard = NA_real_,
        sorensen = NA_real_,
        frac_unique_1 = NA_real_,
        frac_unique_2 = NA_real_
      )
    )
  }
  
  tryCatch(
    {
      hv_set <- hypervolume::hypervolume_set(
        hv1,
        hv2,
        num.points.max = num.points.max.set,
        check.memory = FALSE,
        verbose = FALSE
      )
      
      st <- hypervolume::hypervolume_overlap_statistics(hv_set)
      
      tibble::tibble(
        jaccard = unname(st["jaccard"]),
        sorensen = unname(st["sorensen"]),
        frac_unique_1 = unname(st["frac_unique_1"]),
        frac_unique_2 = unname(st["frac_unique_2"])
      )
    },
    error = function(e) {
      warning(paste("Observed overlap failed:", e$message))
      
      tibble::tibble(
        jaccard = NA_real_,
        sorensen = NA_real_,
        frac_unique_1 = NA_real_,
        frac_unique_2 = NA_real_
      )
    }
  )
}

bootstrap_hv <- function(hv, boot_name, weights, use_weights = TRUE) {
  if (is.null(hv)) return(NA_character_)
  
  w <- NULL
  
  if (isTRUE(use_weights)) {
    w <- clean_weights(weights)
  }
  
  tryCatch(
    {
      if (is.null(w)) {
        out <- hypervolume::hypervolume_resample(
          name = boot_name,
          hv = hv,
          method = "bootstrap",
          n = n_boot,
          points_per_resample = "sample_size",
          cores = cores,
          verbose = TRUE,
          to_file = TRUE
        )
      } else {
        out <- hypervolume::hypervolume_resample(
          name = boot_name,
          hv = hv,
          method = "weighted bootstrap",
          n = n_boot,
          points_per_resample = "sample_size",
          weights = w,
          cores = cores,
          verbose = TRUE,
          to_file = TRUE
        )
      }
      
      as.character(out)[1]
    },
    error = function(e) {
      warning(paste("Bootstrap failed for", boot_name, ":", e$message))
      NA_character_
    }
  )
}

volume_ci_from_path <- function(path, ci = hv_ci) {
  empty_out <- tibble::tibble(
    volume_boot_mean = NA_real_,
    volume_lwr = NA_real_,
    volume_upr = NA_real_
  )
  
  if (length(path) == 0 || all(is.na(path)) || !dir.exists(path)) {
    return(empty_out)
  }
  
  tryCatch(
    {
      hvs <- hypervolume::to_hv_list(path)
      vols <- as.numeric(hypervolume::get_volume(hvs))
      
      vols <- vols[is.finite(vols)]
      
      if (length(vols) == 0) {
        return(empty_out)
      }
      
      alpha <- (1 - ci) / 2
      
      tibble::tibble(
        volume_boot_mean = mean(vols, na.rm = TRUE),
        volume_lwr = unname(stats::quantile(vols, alpha, na.rm = TRUE)),
        volume_upr = unname(stats::quantile(vols, 1 - alpha, na.rm = TRUE))
      )
    },
    error = function(e) {
      warning(paste("Volume CI failed:", e$message))
      empty_out
    }
  )
}

overlap_ci_from_paths <- function(path1, path2, ci = hv_ci) {
  empty_out <- tibble::tibble(
    jaccard_lwr = NA_real_,
    jaccard_upr = NA_real_,
    sorensen_lwr = NA_real_,
    sorensen_upr = NA_real_,
    frac_unique_1_lwr = NA_real_,
    frac_unique_1_upr = NA_real_,
    frac_unique_2_lwr = NA_real_,
    frac_unique_2_upr = NA_real_
  )
  
  if (
    length(path1) == 0 ||
    length(path2) == 0 ||
    all(is.na(path1)) ||
    all(is.na(path2)) ||
    !dir.exists(path1) ||
    !dir.exists(path2)
  ) {
    return(empty_out)
  }
  
  tryCatch(
    {
      res <- hypervolume::hypervolume_overlap_confidence(
        path1 = path1,
        path2 = path2,
        CI = ci,
        cores = cores
      )
      
      tibble::tibble(
        jaccard_lwr = as.numeric(res[["jaccard"]])[1],
        jaccard_upr = as.numeric(res[["jaccard"]])[2],
        sorensen_lwr = as.numeric(res[["sorensen"]])[1],
        sorensen_upr = as.numeric(res[["sorensen"]])[2],
        frac_unique_1_lwr = as.numeric(res[["frac_unique_1"]])[1],
        frac_unique_1_upr = as.numeric(res[["frac_unique_1"]])[2],
        frac_unique_2_lwr = as.numeric(res[["frac_unique_2"]])[1],
        frac_unique_2_upr = as.numeric(res[["frac_unique_2"]])[2]
      )
    },
    error = function(e) {
      warning(paste("Overlap CI failed:", e$message))
      empty_out
    }
  )
}

hv_randompoints_to_plotdf <- function(
    hv,
    id_i,
    niche_i,
    pca_vis,
    hv_axes,
    season_i = NA_character_
) {
  if (is.null(hv)) return(NULL)
  
  rp <- tryCatch(
    as.data.frame(hv@RandomPoints),
    error = function(e) NULL
  )
  
  if (is.null(rp) || nrow(rp) < 3) {
    return(NULL)
  }
  
  if (nrow(rp) > max_plot_points_per_hv) {
    rp <- rp[sample(seq_len(nrow(rp)), max_plot_points_per_hv), , drop = FALSE]
  }
  
  if (all(hv_axes %in% names(rp))) {
    rp <- rp[, hv_axes, drop = FALSE]
  } else {
    rp <- rp[, seq_len(min(length(hv_axes), ncol(rp))), drop = FALSE]
    
    if (ncol(rp) == length(hv_axes)) {
      names(rp) <- hv_axes
    }
  }
  
  rp <- rp |>
    dplyr::mutate(dplyr::across(dplyr::everything(), as.numeric))
  
  rp <- rp[
    stats::complete.cases(rp) &
      apply(as.matrix(rp), 1, function(z) all(is.finite(z))),
    ,
    drop = FALSE
  ]
  
  if (nrow(rp) < 3) return(NULL)
  
  if (!is.null(pca_vis) && ncol(rp) >= 2) {
    pcs <- tryCatch(
      predict(pca_vis, newdata = rp),
      error = function(e) NULL
    )
    
    if (is.null(pcs)) return(NULL)
    
    out <- data.frame(
      PC1 = pcs[, 1],
      PC2 = pcs[, 2],
      id = as.character(id_i),
      season = as.character(season_i),
      niche_type = as.character(niche_i)
    )
  } else if (ncol(rp) == 1) {
    out <- data.frame(
      PC1 = rp[, 1],
      PC2 = 0,
      id = as.character(id_i),
      season = as.character(season_i),
      niche_type = as.character(niche_i)
    )
  } else {
    out <- data.frame(
      PC1 = rp[, 1],
      PC2 = rp[, 2],
      id = as.character(id_i),
      season = as.character(season_i),
      niche_type = as.character(niche_i)
    )
  }
  
  out |>
    dplyr::filter(is.finite(PC1), is.finite(PC2))
}

################################################################################
# 8. REPEATABILITY OF REALIZED AND AVAILABLE NICHE AXES
################################################################################

if (isTRUE(run_repeatability)) {
  safe_section(
    "repeatability",
    {
      log_msg("Starting repeatability workflow.")
      
      repeatability_groups <- groups
      
      df_used <- df_rsp |>
        dplyr::filter(case_num == 1L)
      
      df_avail <- df_rsp |>
        dplyr::filter(case_num == 0L)
      
      compute_repeatability_group <- function(
    dat,
    vars,
    group_label,
    niche_label,
    nboot = rpt_nboot,
    npermut = rpt_npermut
      ) {
        out <- list()
        
        for (v in vars) {
          log_msg(
            "Repeatability: ",
            niche_label,
            ", group ",
            group_label,
            ", variable ",
            v
          )
          
          if (!v %in% names(dat)) {
            warning(paste("Repeatability skipped missing variable:", v))
            next
          }
          
          if (!is.numeric(dat[[v]])) {
            warning(paste("Repeatability skipped non-numeric variable:", v))
            next
          }
          
          if (length(unique(stats::na.omit(dat[[v]]))) < 2) {
            warning(paste("Repeatability skipped zero-variation variable:", v))
            next
          }
          
          rpt_dat <- dat |>
            dplyr::select(id, dplyr::all_of(v)) |>
            tidyr::drop_na()
          
          if (nrow(rpt_dat) < 10 || length(unique(rpt_dat$id)) < 2) {
            warning(paste("Repeatability skipped insufficient data for variable:", v))
            next
          }
          
          form <- stats::as.formula(paste(v, "~ (1|id)"))
          
          rpt_res <- tryCatch(
            rptR::rpt(
              form,
              grname = "id",
              data = rpt_dat,
              datatype = "Gaussian",
              nboot = nboot,
              npermut = npermut
            ),
            error = function(e) {
              warning(paste("Repeatability failed for", v, ":", e$message))
              NULL
            }
          )
          
          if (is.null(rpt_res)) next
          
          one <- tibble::tibble(
            pca_group = group_label,
            variable = v,
            betweenIndR = as.numeric(rpt_res$R["id"]),
            CI_low = as.numeric(rpt_res$CI_emp["id", "2.5%"]),

            niche_type = niche_label
          )
          
          out[[v]] <- one
          
          save_checkpoint(
            dplyr::bind_rows(out),
            paste0(
              "checkpoint_repeatability_",
              safe_name(niche_label),
              "_",
              safe_name(group_label),
              ".rds"
            )
          )
        }
        
        dplyr::bind_rows(out)
      }
      
      results_used <- purrr::imap_dfr(
        repeatability_groups,
        ~ compute_repeatability_group(
          dat = df_used,
          vars = .x,
          group_label = .y,
          niche_label = "Realized"
        )
      )
      
      results_avail <- purrr::imap_dfr(
        repeatability_groups,

          dat = df_avail,
          vars = .x,
          group_label = .y,
          niche_label = "Available"
        )
      )
      
      repeatability_df <- dplyr::bind_rows(results_used, results_avail)
      
      write_csv_safe(
        repeatability_df,
        file.path(table_dir, "repeatability_realized_available_by_group.csv")
      )
      
      save_rds_safe(
        repeatability_df,
        file.path(object_dir, "repeatability_realized_available_by_group.rds")
      )
      
      repeatability_plot_dat <- repeatability_df |>
        dplyr::filter(
          is.finite(betweenIndR),
          is.finite(CI_low),
          is.finite(CI_high)

          repeatability_plot_dat,

            x = variable,
            y = betweenIndR,
            color = niche_type

          ggplot2::geom_point(
            position = ggplot2::position_dodge(width = 0.45),
            size = 2.5
          ) +
          ggplot2::geom_errorbar(
            ggplot2::aes(ymin = CI_low, ymax = CI_high),
            position = ggplot2::position_dodge(width = 0.45),
            width = 0.2
          ) +
          ggplot2::facet_wrap(~ pca_group, scales = "free_x") +
          ggplot2::coord_cartesian(ylim = c(0, 1)) +
          ggplot2::theme_minimal(base_size = 12) +
          ggplot2::labs(
            title = "Repeatability of realized and available environmental conditions by variable group",
            x = "Environmental variable",
            y = "Between-individual repeatability",
            color = "Niche component"
          ) +
          ggplot2::theme(
            axis.text.x = ggplot2::element_text(angle = 45, hjust = 1),
            legend.position = "bottom",
            panel.grid.minor = ggplot2::element_blank()
          )
        
        print(p_repeat)
        
        ggplot2::ggsave(
          file.path(plot_dir, "repeatability_realized_available_by_group.png"),
          p_repeat,
          width = 12,
          height = 7,
          dpi = 300
        )
      } else {
        warning("No finite repeatability estimates available for plotting.")
      }
      
      log_msg("Finished repeatability workflow.")
    }
  )

    } else {
      df_season <- df_rsp |>
        dplyr::mutate(
          date = suppressWarnings(as.Date(as.character(date))),
          month = lubridate::month(date),
          year = lubridate::year(date),
          season = dplyr::case_when(
            month %in% c(3, 4, 5) ~ "Spring",
            month %in% c(6, 7, 8) ~ "Summer",

            month %in% c(12, 1, 2) ~ "Winter",
            TRUE ~ NA_character_
          ),
          season = factor(season, levels = season_levels)
        ) |>
        dplyr::filter(
          !is.na(id),
          !is.na(case_num),
          !is.na(niche_type),
          !is.na(date),
          !is.na(season)

        dplyr::count(season, niche_type, name = "n_rows") |>
        tidyr::complete(
          season,
          niche_type,
          fill = list(n_rows = 0)

        season_counts,
        file.path(table_dir, "seasonal_counts_by_season_and_niche.csv")

        df_season,

      dplyr::mutate(
        id = factor(id, levels = all_ids),
        season = factor(season, levels = season_levels),
        niche_type = factor(niche_type, levels = niche_levels)
      )
    
    if (nrow(seasonal_all_plot_dat) > 0) {
      p_breadth_all <- ggplot2::ggplot() +
        ggplot2::geom_line(
          data = seasonal_all_plot_dat |>
            dplyr::filter(niche_type == "Potential"),
          ggplot2::aes(
            x = season,
            y = volume_observed,
            group = id,
            colour = id
          ),
          alpha = 0.5,
          linewidth = 0.6
        ) +
        ggplot2::geom_point(
          data = seasonal_all_plot_dat |>
            dplyr::filter(niche_type == "Potential"),
          ggplot2::aes(
            x = season,
            y = volume_observed,
            colour = id
          ),
          alpha = 0.5,
          size = 1.5
        ) +
        ggplot2::geom_line(
          data = seasonal_all_plot_dat |>
            dplyr::filter(niche_type == "Realized"),
          ggplot2::aes(
            x = season,
            y = volume_observed,
            group = id,
            colour = id
          ),
          alpha = 1,
          linewidth = 1.2
        ) +
        ggplot2::geom_point(
          data = seasonal_all_plot_dat |>
            dplyr::filter(niche_type == "Realized"),
          ggplot2::aes(
            x = season,
            y = volume_observed,
            colour = id
          ),
          alpha = 1,
          size = 2
        ) +
        ggplot2::facet_grid(niche_type ~ group, scales = "free_y") +
        ggplot2::scale_colour_manual(values = id_cols, drop = FALSE) +
        ggplot2::theme_minimal(base_size = 12) +
        ggplot2::labs(
          title = "Seasonal change in individual hypervolume breadth",
          subtitle = "Potential = transparent; Realized = bold",
          x = "Season",
          y = "Hypervolume breadth",
          colour = "Individual ID"
        ) +
        ggplot2::theme(
          legend.position = "bottom",
          panel.grid.minor = ggplot2::element_blank()
        )
      
      print(p_breadth_all)
      
      ggplot2::ggsave(
        file.path(season_dir, "seasonal_hypervolume_breadth_all_groups.png"),
        p_breadth_all,
        width = 16,
        height = 8,
        dpi = 300
      )
    } else {
      warning("No finite combined seasonal breadth values available for plotting.")
    }
  }
}

################################################################################
# 10. FACETED SEASONAL HYPERVOLUMES BY INDIVIDUAL
################################################################################

safe_section(
  "faceted seasonal hypervolumes by individual",
  {
    log_msg("Starting faceted seasonal hypervolume plots by individual.")
    
    # IMPORTANT:
    # No new folder is created here.
    # Plots will be saved directly in the existing season_dir folder,
    # which was already created in Section 2.
    
    # If needed, load seasonal hypervolume objects from file
    if (!exists("seasonal_hv_objects_by_group")) {
      seasonal_hv_objects_by_group <- readRDS(
        file.path(object_dir, "seasonal_hypervolumes_all_groups_by_id.rds")
      )
      
      log_msg(
        "Loaded seasonal_hv_objects_by_group from: ",
        file.path(object_dir, "seasonal_hypervolumes_all_groups_by_id.rds")
      )
    }
    
    # If needed, recreate helper objects
    if (!exists("all_ids")) {
      all_ids <- sort(unique(as.character(df_rsp$id)))
    }
    
    if (!exists("id_cols")) {
      id_cols <- setNames(
        grDevices::hcl.colors(length(all_ids), palette = "Dark 3"),
        all_ids
      )
    }
    
    if (!exists("season_levels")) {
      season_levels <- c("Spring", "Summer", "Autumn", "Winter")
    }
    
    if (!exists("niche_levels")) {
      niche_levels <- c("Potential", "Realized")
    }
    
    # Function to make distinguishable shades of the same individual colour
    make_season_shade <- function(col, season_i) {
      rgb_mat <- grDevices::col2rgb(col) / 255
      hsv_mat <- grDevices::rgb2hsv(rgb_mat)
      
      hue_i <- hsv_mat["h", 1]
      
      value_i <- c(
        Spring = 1.00,
        Summer = 0.72,
        Autumn = 0.46,
        Winter = 0.24
      )[[as.character(season_i)]]
      
      saturation_i <- 0.95
      
      grDevices::hsv(
        h = hue_i,
        s = saturation_i,
        v = value_i
      )
    }
    
    # Create one colour for each ID-season combination
    id_season_cols <- unlist(
      lapply(
        names(id_cols),
        function(id_i) {
          cols_i <- vapply(
            season_levels,
            function(season_i) {
              make_season_shade(
                col = id_cols[[id_i]],
                season_i = season_i
              )
            },
            character(1)
          )
          
          names(cols_i) <- paste(id_i, season_levels, sep = "___")
          cols_i
        }
      )
    )
    
    # Transparency values for filled hypervolumes
    hv_alpha_values <- c(
      Potential = 0.30,
      Realized = 0.55
    )
    
    # One plot per ecological group
    for (grp in names(seasonal_hv_objects_by_group)) {
      
      log_msg("Creating faceted seasonal hypervolume plot for group: ", grp)
      
      season_hvs <- seasonal_hv_objects_by_group[[grp]]
      
      metadata_path <- file.path(
        object_dir,
        paste0("seasonal_hypervolume_metadata_", grp, ".rds")
      )
      
      if (!file.exists(metadata_path)) {
        warning(
          paste(
            "Skipping group",
            grp,
            "because metadata file was not found:",
            metadata_path
          )
        )
        next
      }
      
      grp_obj <- readRDS(metadata_path)
      
      hv_axes <- grp_obj$hv_axes
      pca_vis <- grp_obj$pca_vis
      
      # Convert hypervolume random points to PC1-PC2 plotting data
      plot_df <- purrr::pmap_dfr(
        list(
          season_hvs$hv,
          season_hvs$id,
          season_hvs$season,
          season_hvs$niche_type
        ),
        function(hv_i, id_i, season_i, niche_i) {
          hv_randompoints_to_plotdf(
            hv = hv_i,
            id_i = id_i,
            niche_i = niche_i,
            season_i = season_i,
            pca_vis = pca_vis,
            hv_axes = hv_axes
          )
        }
      )
      
      if (is.null(plot_df) || nrow(plot_df) == 0) {
        warning(
          paste(
            "No PCA/random-point plot data available for group:",
            grp
          )
        )
        next
      }
      
      # Create convex hull polygons for each ID-season-niche combination
      hull_df <- plot_df |>
        dplyr::mutate(
          id = factor(id, levels = all_ids),
          season = factor(season, levels = season_levels),
          niche_type = factor(niche_type, levels = niche_levels),
          id_season = paste(as.character(id), as.character(season), sep = "___")
        ) |>
        dplyr::group_by(id, season, niche_type, id_season) |>
        dplyr::group_modify(
          ~ {
            out <- close_polygon(.x)
            if (is.null(out)) return(tibble::tibble())
            out
          }
        ) |>
        dplyr::ungroup() |>
        dplyr::filter(
          is.finite(PC1),
          is.finite(PC2)
        )
      
      if (nrow(hull_df) == 0) {
        warning(
          paste(
            "No valid hull polygons available for group:",
            grp
          )
        )
        next
      }
      
      # Transform the PCA hulls so seasons become x-axis categories.
      # Within each individual facet, PC1 is compressed around the season position.
      # PC2 is scaled for display, so the hypervolume shapes fit neatly in each facet.
      panel_scaling <- hull_df |>
        dplyr::group_by(id) |>
        dplyr::summarise(
          PC1_mid = mean(range(PC1, na.rm = TRUE)),
          PC2_mid = mean(range(PC2, na.rm = TRUE)),
          PC1_range = diff(range(PC1, na.rm = TRUE)),
          PC2_range = diff(range(PC2, na.rm = TRUE)),
          .groups = "drop"
        ) |>
        dplyr::mutate(
          PC1_range = dplyr::if_else(PC1_range <= 0, 1, PC1_range),
          PC2_range = dplyr::if_else(PC2_range <= 0, 1, PC2_range)
        )
      
      hull_plot_df <- hull_df |>
        dplyr::left_join(
          panel_scaling,
          by = "id"
        ) |>
        dplyr::mutate(
          season_num = as.numeric(season),
          
          # Controls how wide each hypervolume appears around its season.
          # Increase slightly if polygons look too narrow;
          # decrease if they overlap neighbouring seasons.
          plot_x = season_num + ((PC1 - PC1_mid) / PC1_range) * 0.65,
          
          # Display-scaled PC2.
          plot_y = (PC2 - PC2_mid) / PC2_range,
          
          fill_colour = unname(id_season_cols[as.character(id_season)])
        )
      
      n_id_panels <- length(unique(hull_plot_df$id))
      facet_ncol <- min(4, max(1, ceiling(sqrt(n_id_panels))))
      facet_nrow <- ceiling(n_id_panels / facet_ncol)
      
      plot_width <- max(10, min(22, 4.2 * facet_ncol))
      plot_height <- max(8, min(28, 3.8 * facet_nrow))
      
      p_individual_seasons <- ggplot2::ggplot() +
        
        # Potential niche hypervolumes:
        # dashed black outlines
        ggplot2::geom_polygon(
          data = hull_plot_df |>
            dplyr::filter(niche_type == "Potential"),
          ggplot2::aes(
            x = plot_x,
            y = plot_y,
            group = interaction(id, season, niche_type),
            fill = fill_colour,
            colour = "Potential",
            linetype = "Potential"
          ),
          alpha = hv_alpha_values[["Potential"]],
          linewidth = 0.35
        ) +
        
        # Realized niche hypervolumes:
        # solid thin black outlines
        ggplot2::geom_polygon(
          data = hull_plot_df |>
            dplyr::filter(niche_type == "Realized"),
          ggplot2::aes(
            x = plot_x,
            y = plot_y,
            group = interaction(id, season, niche_type),
            fill = fill_colour,
            colour = "Realized",
            linetype = "Realized"
          ),
          alpha = hv_alpha_values[["Realized"]],
          linewidth = 0.35
        ) +
        
        ggplot2::facet_wrap(
          ~ id,
          scales = "free_y",
          ncol = facet_ncol
        ) +
        
        ggplot2::scale_x_continuous(
          breaks = seq_along(season_levels),
          labels = season_levels,
          limits = c(0.45, length(season_levels) + 0.55),
          expand = ggplot2::expansion(mult = c(0.02, 0.02))
        ) +
        
        ggplot2::scale_fill_identity() +
        
        ggplot2::scale_colour_manual(
          name = "Niche type outline",
          values = c(
            Potential = "black",
            Realized = "black"
          ),
          breaks = c("Potential", "Realized")
        ) +
        
        ggplot2::scale_linetype_manual(
          name = "Niche type outline",
          values = c(
            Potential = "dashed",
            Realized = "solid"
          ),
          breaks = c("Potential", "Realized")
        ) +
        
        ggplot2::guides(
          colour = ggplot2::guide_legend(
            override.aes = list(
              fill = NA,
              linewidth = 0.6
            )
          ),
          linetype = ggplot2::guide_legend(
            override.aes = list(
              fill = NA,
              linewidth = 0.6
            )
          )
        ) +
        
        ggplot2::theme_minimal(base_size = 12) +
        
        ggplot2::labs(
          title = paste("Seasonal hypervolumes by individual -", grp),
          subtitle = paste0(
            "Each facet is one individual. ",
            "Seasons are shown on the x-axis. ",
            "Colours are season-specific shades of each individual's colour. ",
            "Dashed outlines = Potential niche; solid outlines = Realized niche."
          ),
          x = "Season",
          y = "Scaled PC2 projection"
        ) +
        
        ggplot2::theme(
          legend.position = "bottom",
          panel.grid.minor = ggplot2::element_blank(),
          panel.grid.major.x = ggplot2::element_blank(),
          axis.text.x = ggplot2::element_text(angle = 30, hjust = 1),
          strip.text = ggplot2::element_text(face = "bold"),
          plot.margin = ggplot2::margin(15, 25, 15, 25)
        )
      
      print(p_individual_seasons)
      
      # Save directly in the existing seasonal hypervolume plot folder: season_dir
      ggplot2::ggsave(
        filename = file.path(
          season_dir,
          paste0("seasonal_hypervolume_by_individual_", grp, ".png")
        ),
        plot = p_individual_seasons,
        width = plot_width,
        height = plot_height,
        dpi = 300
      )
      
      ggplot2::ggsave(
        filename = file.path(
          season_dir,
          paste0("seasonal_hypervolume_by_individual_", grp, ".pdf")
        ),
        plot = p_individual_seasons,
        width = plot_width,
        height = plot_height
      )
      
      log_msg(
        "Saved faceted seasonal hypervolume plot for group: ",
        grp,
        " to existing folder: ",
        season_dir
      )
    }
    
    log_msg("Finished faceted seasonal hypervolume plots by individual.")
  }
)
