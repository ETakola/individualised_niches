# fixed slope helper (from your *_fix tables)
get_beta <- function(fix_tbl, term_name) {
  fix_tbl %>% filter(term == term_name) %>% pull(estimate) %>% .[1]
}

# x-seq helper (from your data)
make_xseq <- function(dat, var, n = 60) {
  seq(min(dat[[var]], na.rm = TRUE),
      max(dat[[var]], na.rm = TRUE),
      length.out = n)
}

# build straight-line reaction norms for one slope object
build_rn_lines <- function(slopes_tbl, fix_tbl, term_name, dat, xvec, model_label, term_label, link = c("identity","logit")) {
  link <- match.arg(link)
  beta <- get_beta(fix_tbl, term_name)
  
  slopes_tbl %>%
    mutate(
      model = model_label,
      term  = term_label,
      slope_total = beta + slope_blup
    ) %>%
    crossing(x = xvec) %>%
    mutate(
      eta = slope_total * x,                     # straight line on link/identity scale
      y   = if (link == "logit") eta else eta    # keep as eta to guarantee straight lines
    ) %>%
    dplyr::select(model, term, id, x, y)
}

# Relative selection probabilities / relative selection strength
################################################################################ 
# This is individual-specific relative selection strength from the fitted RSF. 

eta <- predict( # Predict selection score for every location.
  rsf, # Use fitted RSF model.
  newdata = df, # Predict for the original used/available dataset.
  type = "link", # Predict on logit scale.
  re.form = NULL # Include individual random effects.
)

df_rsp <- df %>% 
  dplyr::mutate( # Add prediction column to the dataset.
    eta = as.numeric(eta) # Store linear predictor as numeric.
  ) %>% 
  dplyr::group_by(id) %>% # Scale within each individual.
  dplyr::mutate( # Add scaled RSP.
    RSP_scaled = exp(eta - max(eta, na.rm = TRUE)) # Stable exponentiation; max RSP per individual is 1.
  ) %>% # Continue pipeline.
  dplyr::ungroup() %>% # Remove grouping.
  dplyr::mutate( # Add niche-component label.
    niche_type = dplyr::if_else(case_ == 1, "Realized", "Available") # Label used vs available locations.
  ) 

write.csv(df_rsp,

    n = dplyr::n(), # Count rows.
    mean_RSP_scaled = mean(RSP_scaled, na.rm = TRUE), # Mean RSP.
